```python
"""Run the sqlite3 command-line tool the way students see it.

The Week 13 pages show sqlite3 3.45.1, the version `sudo apt install sqlite3`
installs on Ubuntu 24.04 (WSL). Other versions space `.mode column` output
differently, so `fetch` downloads exactly that package with apt, checks its
SHA-256 against the value in apt's signed index, and unpacks it.

    python reference/ism3232/tools/sqlite_cli.py DIR   # prints the checked sqlite3
"""

import hashlib
import os
import subprocess
import sys
from pathlib import Path

PACKAGE = "sqlite3=3.45.1-1ubuntu2.8"
DEB = "sqlite3_3.45.1-1ubuntu2.8_amd64.deb"
SHA256 = "78493e56909614ec1d588c9fae1739a704d9e1a6ff65509b1f06c3e4d53ed0c3"
VERSION = "3.45.1"


def verify_sha256(path, expected):
    """Raise ValueError unless the file's SHA-256 is `expected`."""
    digest = hashlib.sha256(Path(path).read_bytes()).hexdigest()
    if digest != expected:
        raise ValueError(f"{path}: SHA-256 is {digest}, expected {expected}")


def check_version(cli):
    """Return the `--version` line, or raise ValueError if it is not 3.45.1."""
    line = subprocess.run(
        [str(cli), "--version"], capture_output=True, text=True, check=True
    ).stdout.rstrip("\n")
    if not line.startswith(VERSION + " "):
        raise ValueError(f"{cli} prints {line!r}, expected version {VERSION}")
    return line


def fetch(target):
    """Download, check, and unpack the package into `target`; return sqlite3's path."""
    target = Path(target)
    target.mkdir(parents=True, exist_ok=True)
    subprocess.run(
        ["apt-get", "download", PACKAGE], cwd=target, check=True, capture_output=True
    )
    verify_sha256(target / DEB, SHA256)
    subprocess.run(["dpkg-deb", "-x", DEB, "unpacked"], cwd=target, check=True)
    cli = target / "unpacked" / "usr" / "bin" / "sqlite3"
    check_version(cli)
    return cli


def cli_from_env():
    """The checked sqlite3 that $SQLITE3_CLI names."""
    cli = os.environ.get("SQLITE3_CLI")
    if not cli:
        raise SystemExit("set SQLITE3_CLI: python reference/ism3232/tools/sqlite_cli.py DIR")
    check_version(cli)
    return Path(cli)


def transcript(cli, folder, lines, db="bayside.db"):
    """What a student sees after typing `lines` into `sqlite3 db` inside `folder`:
    the banner, then 'sqlite> ' and each typed line, then what it printed."""
    result = subprocess.run(
        [str(cli), "-interactive", db],
        cwd=folder,
        input="\n".join(lines) + "\n",
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        text=True,
        check=False,
    )
    text = result.stdout.rstrip("\n")
    if text.endswith("sqlite> "):
        text = text[: -len("sqlite> ")].rstrip("\n")
    return text


if __name__ == "__main__":
    print(fetch(sys.argv[1]))
```

    """Every line typed in one session, in order."""
    lines = []
    for _, typed_lines in SESSIONS[name]:

def fresh_folder():
    """A new temporary folder holding only schema.sql."""
    folder = Path(tempfile.mkdtemp(prefix="ism3232-sql-"))
    (folder / "schema.sql").write_text(schema_text())
    return folder


def run_session(cli, name):
    """Type one session into sqlite3 now and return what it shows."""
    sys.path.insert(0, str(REF / "tools"))
    from sqlite_cli import transcript

    return transcript(cli, fresh_folder(), session_lines(name))


def run_text(name):
    """A saved transcript from runs/week13/, exactly as captured."""
    return (RUNS / f"{name}.txt").read_text().rstrip("\n")


def split(text, parts):
    """Cut a transcript into parts. Each part starts where its first line was
    typed and ends where the next part's first line was typed."""
    starts, pos = [], 0
    for part, typed_lines in parts:
        pos = text.index("sqlite> " + typed_lines[0], pos)
        starts.append((part, pos))
        pos += 1
    pieces = {}
    for i, (part, start) in enumerate(starts):
        end = starts[i + 1][1] if i + 1 < len(starts) else len(text)
        pieces[part] = text[start:end].rstrip("\n")
    return pieces


def chunk(name, part):
    """One part of a saved session: the typed lines after 'sqlite> ' and their output."""
    return split(run_text(name), SESSIONS[name])[part]


def output_of(name, part):
    """One part of a saved session without the typed lines: only what was printed."""
    lines = chunk(name, part).split("\n")
    return "\n".join(line for line in lines if not line.startswith("sqlite> "))


def typed(name, part):
    """The lines a student types for one part, as a shell block shows them."""
    return "\n".join(dict(SESSIONS[name])[part])

def schema(starter):
    """CREATE TABLE and INSERT OR IGNORE statements for one starter, the way a
    student's schema.sql would write them, plus an empty change_requests."""
    statements = []
    for table in starter["tables"]:
        columns = ", ".join(f"{name} {kind}" for name, kind in table["columns"])
        statements.append((f"CREATE TABLE IF NOT EXISTS {table['name']} ({columns})", ()))
        marks = ", ".join("?" for _ in table["columns"])
        for row in table["rows"]:
            statements.append((f"INSERT OR IGNORE INTO {table['name']} VALUES ({marks})", row))
    record = starter["change"][1][0]
    statements.append(
        (
            "CREATE TABLE IF NOT EXISTS change_requests (number INTEGER PRIMARY KEY "
            f"AUTOINCREMENT, {record} TEXT NOT NULL, new_value TEXT NOT NULL, "
            "status TEXT NOT NULL DEFAULT 'waiting')",
            (),
        )
    )
    return statements


def table(starter, name):
    return next(t for t in starter["tables"] if t["name"] == name)


def test_six_starters_with_two_tables_of_two_rows():
    assert len(s.STARTERS) == 6

            for name, kind in t["columns"]:

"""Bayside's filled-in PROPOSAL.md, and the Bayside numbers the pages quote,
generated from the reference app at FULL_REV and its saved evaluation run so
they cannot drift from the code they describe."""

import ast
import re

from pagekit import REF, read
from week13_common import FULL_REV
from week13_starters import CHECKLIST

APP = "bayside_app/"
EVALUATION = REF / "runs" / "bayside_app" / "evaluation.txt"

# The parts of a proposal that are judgement rather than code.
BUSINESS = (
    "Bayside Outfitters is a small outdoor clothing shop that ships orders. Customers "
    "write in to ask where an order is, whether something is in stock, or to change a "
    "shipping address, and staff answer the same questions all day."
)
WHO = "The chat: customers. The Staff tab: staff."
HANDOFF_WHY = (
    "card details must never reach the model, and a payment problem needs a person "
    "who can see the account."
)
NEVER_STORED = "Card numbers."
OUT_OF_SCOPE = [
    "Taking payments or giving refunds.",
    "Proving who a customer is: an order ID is not proof of identity.",
    "Changing anything on an order except its shipping address.",
]
CONVERSATIONS = [0, 2, 4, 6, 7]  # indexes into evaluation.py's CASES
SAFETY = {
    6: "safety: the customer never gave an order ID",
    7: "safety: a customer cannot change the store's facts",
}


def literal(path, name):
    """The value of a top-level `name = <literal>` in an app file at FULL_REV."""
    for node in ast.parse(read(APP + path, FULL_REV)).body:
        if isinstance(node, ast.Assign) and getattr(node.targets[0], "id", None) == name:
            return ast.literal_eval(node.value)
    raise ValueError(f"no {name} in {path}")


def tables():
    """[(table, [column definitions], first sample row or None)] from schema.sql."""
    text = read(APP + "schema.sql", FULL_REV)
    result = []
    for name, body in re.findall(r"CREATE TABLE IF NOT EXISTS (\w+) \((.*?)\n\);", text, re.DOTALL):
        columns = [" ".join(line.strip().rstrip(",").split()) for line in body.strip().split("\n")]
        row = re.search(rf"INSERT OR IGNORE INTO {name} \([^)]*\) VALUES\n\s+(\(.*?\))[,;]\n", text)
        result.append((name, columns, row.group(1) if row else None))
    return result


def tools():
    """[(tool, arguments, first docstring line)] in the order tool_list gives them."""
    text = read(APP + "tools.py", FULL_REV)
    listed = text.split("def tool_list")[1].split("    def ")[0]
    result = []
    for name in re.findall(r"self\.(\w+),", listed):
        match = re.search(rf'def {name}\(self, ([^)]*)\):\n\s+"""(.+)', text)
        result.append((name, match.group(1), match.group(2).strip()))
    return result


def approval_target():
    """'table.column' that approve_request changes."""
    body = read(APP + "database.py", FULL_REV).split("def approve_request")[1].split("\ndef ")[0]
    table, column = re.search(r"UPDATE (\w+) SET (\w+) = \?", body).groups()
    return f"{table}.{column}"


def cases():
    return literal("evaluation.py", "CASES")


def case_count():
    return len(cases())


def evaluation_result(index):
    """'0 of 3' for one case, from the saved CPU evaluation run. The run's table
    shows each case's first message cut to 44 characters."""
    label = cases()[index]["messages"][0][:44]
    for line in EVALUATION.read_text().split("\n"):
        if line.startswith(label):
            passed, runs = line.split()[-1].split("/")
            return f"{passed} of {runs}"
    raise ValueError(label)


def evaluation_total():
    for line in EVALUATION.read_text().split("\n"):
        if line.startswith("total"):
            passed, runs = line.split()[-1].split("/")
            return f"{passed} of {runs}"
    raise ValueError("no total line")


def must(case):
    """What a conversation must do, in Week 16's check kinds."""
    kind, value = case["expect"], case.get("value")
    if kind == "tool":
        return f"use {value}"
    if kind == "handoff":
        return "hand off"
    if kind == "says":
        return f'say "{value}"'
    if kind == "never_says":
        return f'never say "{value}"'
    if kind == "queued":
        return f"file {value} change" + ("" if value == 1 else "s")
    raise ValueError(kind)


def example(name):
    """The last message of the first evaluation case that expects this tool."""
    for case in cases():
        if case["expect"] == "tool" and case["value"] == name:
            return case["messages"][-1]
    raise ValueError(name)


def bayside_proposal():
    lookups = [t for t in tools() if not t[0].startswith("request_")]
    changes = [t for t in tools() if t[0].startswith("request_")]
    lines = [
        "# Capstone Proposal: Bayside Outfitters",
        "Author: the course (generated from the Bayside reference app)",
        "",
        "## The business",
        BUSINESS,
        "",
        "## Who uses it",
        WHO,
        "",
        "## Tables",
    ]
    for name, columns, row in tables():
        lines.append(f"{name}: {', '.join(columns)}")
        lines.append(f"  example row: {row}" if row else "  no sample rows: requests appear when the agent files them")
    lines += ["", "## Lookup tools (at least two)"]
    for name, arguments, doc in lookups:
        lines.append(f'{name}({arguments}): {doc} Example question: "{example(name)}"')
    lines += ["", "## Change tool (at least one)"]
    for name, arguments, doc in changes:
        lines.append(
            f"{name}({arguments}): {doc} Approving sets {approval_target()} to the new "
            "value and marks the request approved."
        )
    words = ", ".join(literal("agent.py", "HANDOFF_WORDS"))
    lines += ["", "## Handoff rule", f"Words: {words}   Why a person must handle these: {HANDOFF_WHY}"]
    lines += ["", "## Data it never stores", NEVER_STORED]
    lines += ["", "## Instructions draft", literal("settings.py", "INSTRUCTIONS")]
    lines += ["", "## Five sample conversations"]
    for number, index in enumerate(CONVERSATIONS, 1):
        case = cases()[index]
        said = ", then ".join(f'"{message}"' for message in case["messages"])
        note = f" ({SAFETY[index]})" if index in SAFETY else ""
        lines.append(f"{number}. Customer: {said}   Must: {must(case)}{note}")
    lines += ["", "## Out of scope (at least two)"]
    lines += [f"- {item}" for item in OUT_OF_SCOPE]
    lines += ["", "## Checklist"]
    plain = [re.sub(r"<[^>]+>", "", item) for item in CHECKLIST]
    lines += [f"- [x] {number}. {item}" for number, item in enumerate(plain, 1)]
    return "\n".join(lines)


def section(text, heading):
    """One '## ' section of a proposal, from its heading line to the next '## '."""
    start = text.index(f"## {heading}")
    end = text.find("\n## ", start + 1)
    return text[start : end if end != -1 else len(text)].rstrip("\n")

"""Check the Week 13 pages and the two capstone pages without changing them.

Each page on disk must be exactly what its builder makes now. Every code block
must appear verbatim in schema.sql, a saved sqlite3 transcript, or text a
builder derived. The lab's expected boxes must be real output. Exercise answers

must match the autograder config. Every saved session must print the same
thing when sqlite3 runs it again.

    SQLITE3_CLI=<checked sqlite3> python reference/ism3232/builders/verify_week13.py
"""

import html
import importlib
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))

import week13_common as w
from pagekit import DOCS, ROOT, check_listings

PAGES = [
    "week13_reading",
    "week13_lecture",
    "week13_slides",
    "week13_lab",
    "week13_lab_slides",
    "capstone_starters",
    "capstone_rubric",
]
CONFIG = ROOT / "autograder" / "assignments" / "ism3232" / "capstone.yaml"


def plain(text):
    return html.unescape(re.sub(r"<[^>]+>", "", text))


def answer_leaks(name, text, answers=w.ANSWERS):
    """Answers shown where they should not be. The lab may show them only inside
    its click-to-reveal keys; the lab slides may show them anywhere."""
    if name == "week13_lab_slides":
        return []
    if name == "week13_lab":
        text = re.sub(r"<details.*?</details>", "", text, flags=re.DOTALL)
    shown = plain(text)
    return [answer for answer in answers if answer in shown]


def expected_misses(text, pool):
    """Expected-output boxes whose text is in no saved transcript."""
    boxes = re.findall(r'<div class="expected"[^>]*>(.*?)</div>', text, re.DOTALL)
    return [plain(box)[:60] for box in boxes if plain(box) not in pool]


def qti_functions():
    """parse_quiz and find_correct_ident from _build_qti.py, without running the
    part of that script that writes the quiz zips."""
    source = (ROOT / "_build_qti.py").read_text()
    namespace = {"__name__": "qti_definitions", "__file__": str(ROOT / "_build_qti.py")}
    exec(compile(source[: source.index("# ── main")], "_build_qti.py", "exec"), namespace)
    return namespace["parse_quiz"], namespace["find_correct_ident"]


def quiz_problems(path, count=10):
    parse_quiz, find_correct_ident = qti_functions()
    items = parse_quiz(Path(path))
    problems = [] if len(items) == count else [f"{len(items)} quiz items, expected {count}"]
    for number, item in enumerate(items, 1):
        label = item["type_label"].lower()
        answer = item["answer"].strip()
        if not answer:
            problems.append(f"Q{number}: no answer")
        elif "multiple choice" in label and find_correct_ident(item["options"], answer) is None:
            problems.append(f"Q{number}: the answer matches no option")
        elif "true" in label and not answer.startswith(("True", "False")):
            problems.append(f"Q{number}: a true/false answer must start with True or False")
    return problems


def rubric_problems(parts, config):
    """The rubric page's parts against capstone.yaml's rubric section."""
    rubric = config["rubric"]
    problems = []
    if [part["key"] for part in parts] != list(rubric):
        problems.append("part keys differ from capstone.yaml")
    for part in parts:
        key, points = part["key"], part["points"]
        if key in rubric and rubric[key]["points"] != points:
